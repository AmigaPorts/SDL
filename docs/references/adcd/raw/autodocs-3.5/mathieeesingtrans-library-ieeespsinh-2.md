# mathieeesingtrans.library/IEEESPSinh



   NAME
	IEEESPSinh -- compute the hyperbolic sine of a floating point number

   SYNOPSIS
	  x   = IEEESPSinh(  y  );
	d0		   d0

	float	x,y;

   FUNCTION
	Compute hyperbolic sine of y in IEEE single precision

   INPUTS
	y - IEEE single precision floating point value
